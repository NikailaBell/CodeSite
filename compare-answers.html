Compare with an Expert

A scenario-based exercise where learners tackle a real-world problem and immediately compare their response side-by-side with an expert's to see how closely their thinking aligns.

🛠️ How to Add or Remove Questions

If you have more or fewer questions than this template, just look at the code and edit to the number you need. Copy the <article class="cp-card"> section and change the number label (e.g., q3) to add questions, or delete an article to remove one.

Important: If you add questions, scroll to the bottom of the code and add your new q3 or q4 labels to the taIds and lsKeys variables in the JavaScript!

Q2: "Let's delay to Tuesday. I will run the videos through auto-captioning and edit them. The developer can add basic tab-index navigation."

  <!-- JavaScript Logic -->
  <script>
    (function(){
      // IMPORTANT: If you add questions, add "q3", "q4" to BOTH arrays below!
      const taIds = ["q1", "q2"];
      const lsKeys = {
        q1: "activity_autosave_q1",
        q2: "activity_autosave_q2"
      };
      
      const textareas = taIds.map(id => document.getElementById(id));
      const btnCompare = document.getElementById('btnCompare');
      const btnEdit = document.getElementById('btnEdit');
      const btnExport = document.getElementById('btnExport');
      const toast = document.getElementById('cp-toast');
      const questionsSec = document.getElementById('cp-questions');
      const compareSec = document.getElementById('cp-compare');
      const buttonsDiv = document.getElementById('cp-buttons');
      const expertScroll = document.getElementById('expert-scroll');

      function autogrow(el){ if(!el) return; el.style.height = 'auto'; el.style.height = (el.scrollHeight) + 'px'; }

      // Restore autosave
      textareas.forEach(ta => {
        if(localStorage.getItem(lsKeys[ta.id])) ta.value = localStorage.getItem(lsKeys[ta.id]);
        autogrow(ta);
        ta.addEventListener('input', () => {
          autogrow(ta);
          localStorage.setItem(lsKeys[ta.id], ta.value);
        });
      });

      // Compare Mode
      btnCompare.addEventListener('click', () => {
        textareas.forEach(ta => ta.setAttribute('readonly','true'));
        taIds.forEach(id => {
          const uAns = document.getElementById('user-' + id);
          if(uAns) uAns.textContent = document.getElementById(id).value || "(No answer provided.)";
        });
        
        questionsSec.style.display = 'none';
        compareSec.style.display = 'block';
        btnCompare.classList.add('cp-hidden');
        btnEdit.classList.remove('cp-hidden');
        
        // Move buttons below compare section
        compareSec.appendChild(buttonsDiv);
        
        // Match heights
        const userBody = document.querySelector('.cp-col-body');
        expertScroll.style.maxHeight = Math.max(userBody.scrollHeight, 150) + 'px';
        expertScroll.style.overflowY = 'auto';
      });

      // Edit Mode
      btnEdit.addEventListener('click', () => {
        textareas.forEach(ta => ta.removeAttribute('readonly'));
        compareSec.style.display = 'none';
        questionsSec.style.display = 'block';
        btnEdit.classList.add('cp-hidden');
        btnCompare.classList.remove('cp-hidden');
        questionsSec.appendChild(buttonsDiv);
      });

      // Export
      btnExport.addEventListener('click', async () => {
        let text = "My Practice Answers:\n\n";
        taIds.forEach((id, index) => {
          text += `Q${index+1}:\n${document.getElementById(id).value || "(No answer)"}\n\n`;
        });
        try {
          await navigator.clipboard.writeText(text);
          toast.textContent = "Copied to clipboard!";
        } catch(e) {
          toast.textContent = "Failed to copy.";
        }
        setTimeout(() => toast.textContent = "", 2500);
      });
    })();
  </script>
